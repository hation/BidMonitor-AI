```python
"""AI deep-read via local deerflow CLI: config, prompt, run, parse, hash."""
import hashlib
import json
import os
import re
import shutil
import subprocess
import tempfile
from pathlib import Path


DEFAULT_AI_SETTINGS = {
    "enabled": False,
    "deerflow_cli": "",
    "gateway": "http://127.0.0.1:8001",
    "timeout": 600,
    "recursion_limit": 300,
    "detail_len": 2000,
    "auto_enabled": False,
    "auto_limit": 10,
}


def load_ai_config(store):
    return DEFAULT_AI_SETTINGS | (store.setting('ai_read_settings') or {})


def locate_deerflow_cli(conf):
    explicit = (conf.get('deerflow_cli') or '').strip()
    if explicit:
        return explicit
    found = shutil.which('deerflowcli')
    if found:
        return found
    for candidate in (Path.home() / '.local' / 'bin', Path.home() / 'bin',
                      Path('/usr/local/bin'), Path('/opt/homebrew/bin')):
        path = candidate / 'deerflowcli'
        if path.exists() and os.access(path, os.X_OK):
            return str(path)
    return ''


def build_prompt(project, conf):
    detail_len = int(conf.get('detail_len', 2000))
    fields = {
        '预算': project.get('budget', ''), '投标截止': project.get('deadline', ''),
        '采购人': project.get('purchaser', ''), '中标单位': project.get('win_company', ''),
        '中标金额': project.get('win_amount', ''),
    }
    field_lines = '\n'.join(f'{k}：{v}' for k, v in fields.items() if v)
    content = (project.get('content') or '')[:detail_len]
    return (
        f"请精读以下招标公告，输出 JSON（字段：采购单位/预算/投标截止/技术要点/业务关联度/跟进建议/风险提示），"
        f"并附一段『建议文本』给出可落地的跟进建议。可 web_fetch 打开原文链接核实。\n\n"
        f"标题：{project.get('title', '')}\n{field_lines}\n正文：{content}"
    )


def run_deerflow(task, conf):
    """Execute deerflowcli; return the produced report text or an error dict."""
    cli = locate_deerflow_cli(conf)
    if not cli:
        return {'error': '未配置 deerflow CLI 路径；请在设置页填写，或安装 deerflowcli 到 PATH'}
    out_dir = tempfile.mkdtemp(prefix='ai_read_')
    cmd = [cli, '--output-dir', out_dir, '--timeout', str(int(conf.get('timeout', 600))), '--no-stream']
    gateway = (conf.get('gateway') or '').strip()
    if gateway and gateway != 'http://127.0.0.1:8001':
        cmd += ['--gateway', gateway]
    recursion = int(conf.get('recursion_limit', 300))
    if recursion:
        cmd += ['--recursion-limit', str(recursion)]
    cmd.append(task)
    try:
        proc = subprocess.run(cmd, capture_output=True, text=True, timeout=int(conf.get('timeout', 600)) + 60)
    except subprocess.TimeoutExpired:
        return {'error': 'AI 精读超时，请稍后重试'}
    except Exception as error:
        return {'error': f'deerflow 执行失败：{type(error).__name__}'}
    if proc.returncode != 0:
        text = (proc.stderr or proc.stdout or '')
        if re.search(r'(api[ -]?key|unauthorized|401|403|invalid[ _-]?credential)', text, re.I):
            return {'error': '模型凭据无效或已过期，请检查 deerflow Gateway 配置'}
        if re.search(r'(rate[ _-]?limit|429|quota|insufficient)', text, re.I):
            return {'error': '模型限流，请稍后重试'}
        if re.search(r'(connection|refused|timeout|unreachable|gateway)', text, re.I):
            return {'error': 'deerflow Gateway 不可达，请确认已启动（curl http://127.0.0.1:8001/health）'}
        return {'error': f'deerflow 退出码 {proc.returncode}'}
    hits = []
    for root, _, files in os.walk(out_dir):
        for fn in files:
            if fn.endswith('.md'):
                hits.append(Path(root) / fn)
    if not hits:
        return {'error': 'deerflow 未产出报告文件，请查看 deerflow 日志'}
    report = max(hits, key=os.path.getmtime).read_text(encoding='utf-8', errors='replace')
    return report


def parse_result(text):
    """Extract JSON from model output; degrade to an error dict on failure."""
    if not text:
        return {'error': 'AI 输出为空'}
    content = re.sub(r'^```(?:json)?\s*', '', text.strip(), flags=re.I)
    content = re.sub(r'\s*```$', '', content).strip()
    start, end = content.find('{'), content.rfind('}')
    if start < 0 or end <= start:
        return {'error': 'AI 输出解析失败', 'raw': text[:500]}
    try:
        return json.loads(content[start:end + 1])
    except ValueError:
        return {'error': 'AI 输出解析失败', 'raw': text[:500]}


def compute_hash(project, conf):
    detail_len = int(conf.get('detail_len', 2000))
    source = '|'.join([
        project.get('title', ''), project.get('budget', ''), project.get('deadline', ''),
        project.get('purchaser', ''), project.get('win_company', ''), project.get('win_amount', ''),
        (project.get('content') or '')[:detail_len],
    ])
    return hashlib.md5(source.encode('utf-8')).hexdigest()
```

```python
class AIReadFetcher:
    """Background AI deep-read: manual queue first, pending/changed scans, 60s pacing."""

    def __init__(self, store, runner=run_deerflow, interval=60):
        self.store = store
        self.runner = runner
        self.interval = max(1, interval)
        self.stop = threading.Event()
        self.thread = None
        self.queue = queue.Queue()
        self.job = dict(running=False, message='等待精读', completed=0, total=0, failures=0)

    def request(self, key):
        self.queue.put(key)

    def _snapshot(self):
        with self.store.connection() as db:
            db.execute('INSERT INTO workbench_settings(key,value) VALUES (?,?) '
                       'ON CONFLICT(key) DO UPDATE SET value=excluded.value',
                       ('ai_read_job', __import__('json').dumps(self.job, ensure_ascii=False)))

    def maybe_schedule(self):
        """Auto-schedule after collection: queue direct-relevance notices without a current read."""
        conf = load_ai_config(self.store)
        if not conf.get('enabled') or not conf.get('auto_enabled'):
            return
        limit = int(conf.get('auto_limit', 10))
        rows = self.store.search(relevance='direct', limit=50)['items']
        queued = 0
        for row in rows:
            if queued >= limit:
                break
            project = self.store.project(row['id'])
            if not project:
                continue
            current_hash = compute_hash(project, conf)
            existing = self.store.ai_read(row['id'])
            if existing and existing['status'] == 'ready' and existing['input_hash'] == current_hash:
                continue
            self.store.save_ai_read(row['id'], 'pending', input_hash=current_hash)
            self.request(row['id'])
            queued += 1

    def run_once(self):
        tasks = []
        while True:
            try:
                tasks.append(self.queue.get_nowait())
            except queue.Empty:
                break
        conf = load_ai_config(self.store)
        if conf.get('enabled'):
            with self.store.connection() as db:
                for project_id, in db.execute(
                        "SELECT project_id FROM ai_reads WHERE status='pending' "
                        "ORDER BY created_at ASC LIMIT 50"):
                    tasks.append(project_id)
        seen = set()
        for key in tasks:
            if self.stop.is_set() or key in seen:
                continue
            seen.add(key)
            project = self.store.project(key)
            if not project or not project.get('url'):
                continue
            current_hash = compute_hash(project, conf)
            existing = self.store.ai_read(key)
            if existing and existing['status'] == 'ready' and existing['input_hash'] == current_hash:
                continue
            self.store.save_ai_read(key, 'running', input_hash=current_hash)
            try:
                prompt = build_prompt(project, conf)
                outcome = self.runner(prompt, conf)
                if isinstance(outcome, dict) and outcome.get('error'):
                    self.store.save_ai_read(key, 'error', result={}, error=outcome['error'], input_hash=current_hash)
                else:
                    result = parse_result(outcome)
                    status = 'ready' if not result.get('error') else 'error'
                    error = result.get('error', '') if status == 'error' else ''
                    self.store.save_ai_read(key, status, result=result, error=error, input_hash=current_hash)
            except Exception as error:
                self.store.save_ai_read(key, 'error', result={}, error=f'精读异常：{type(error).__name__}', input_hash=current_hash)
            self.stop.wait(self.interval)

    def _loop(self):
        while not self.stop.is_set():
            try:
                self.run_once()
            except Exception as error:
                pass
            self.stop.wait(self.interval)

    def open(self):
        if self.thread and self.thread.is_alive():
            return
        self.stop.clear()
        self.thread = threading.Thread(target=self._loop, daemon=True, name='workbench-ai-read')
        self.thread.start()

    def close(self):
        self.stop.set()
        if self.thread:
            self.thread.join(timeout=2)
```

> 说明：`AIReadFetcher` 用独立 `self.stop`（与采集/详情补抓分离）。`run_once` 手动队列优先；`enabled` 时才扫 `ai_reads` pending。`ready+hash 一致` 跳过（缓存）。`parse_result` 失败 → status='error'（result 含 error+raw）。`maybe_schedule()` 是自动精读入口：`auto_enabled` 时对 `relevance=direct` 且无当前结果（或 hash 变化）的公告写 pending 并入队，受 `auto_limit` 限制。

function renderAiRead(p) {
  const box = $('#aiReadResult'); if (!box) return;
  const record = p.ai_read;
  if (!record || record.status === 'pending' || record.status === 'running') {
    box.innerHTML = '<div class="muted">精读中…（后台执行，可能需数分钟）</div>';
    $('#aiReadButton').disabled = true;
    if (p.id) setTimeout(() => pollAiRead(p.id), 8000);
    return;
  }
  $('#aiReadButton').disabled = false;
  if (record.status === 'error') { box.innerHTML = `<div class="error-box">${esc(record.error || '精读失败')}</div>`; return; }
  const r = record.result || {};
  const items = [['采购单位', r.采购单位], ['预算', r.预算], ['投标截止', r.投标截止], ['技术要点', r.技术要点], ['业务关联度', r.业务关联度]].filter(([,v]) => v);
  box.innerHTML = items.map(([k,v]) => `<div class="info-item"><small>${k}</small><span>${esc(String(v))}</span></div>`).join('') +
    (r.跟进建议 ? `<div class="info-item"><small>跟进建议</small><span>${esc(String(r.跟进建议))}</span></div>` : '') +
    (r.风险提示 ? `<div class="info-item"><small>风险提示</small><span>${esc(String(r.风险提示))}</span></div>` : '') +
    (r.建议文本 ? `<p class="muted" style="margin-top:12px">${esc(String(r.建议文本))}</p>` : '');
}
