# 公告详情与中标/废标追踪 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 工作台后台异步补抓公告详情正文并提取金额/截止/中标单位等结构化字段，同时放开结果公告过滤，让中标/废标公告入库、中标单位/金额可见可筛。

**Architecture:** 全部在现有 `apps/workbench/bidmonitor_workbench` 内实现，SQLite 加列走现有 ALTER TABLE 兼容模式。详情抓取复用已就绪的 `collectors/detail.py::fetch_detail/extract_fields`；新增 CollectionService 内后台补抓线程（令牌桶限速 + host 冷却）；中标/废标过滤在采集（generic/feeds）与搜索（search）三处放开；中标单位/金额用正则从正文提取。

**Tech Stack:** Python 3.10+ · FastAPI · SQLite（标准库 sqlite3，WAL，无 ORM/迁移框架）· pytest（`tests/`）· 原生 JS 前端（无框架）

在 `CollectionService` 类定义**之前**新增 `DetailFetcher`：

```python
class DetailFetcher:
    """Background detail backfill: bounded scans, token-bucket pacing, host cooldowns."""

    def __init__(self, store, fetch=fetch_detail, rate_per_min=20):
        self.store = store
        self.fetch = fetch
        self.rate_per_min = rate_per_min
        self.interval = max(1.0, 60.0 / rate_per_min)
        self.stop = threading.Event()
        self.thread = None
        self.queue = queue.Queue()
        self.host_cooldowns = {}

    def request(self, key, url):
        self.queue.put((key, url))

    def _blocked(self, error):
        text = str(error)
        return bool(re.search(r'HTTP (?:401|403|429)|安全验证|验证码|限制访问|限流提示页|访问受限|blocked', text, re.I))

    def run_once(self):
        # Manual queue first, then pending backlog.
        tasks = []
        while True:
            try:
                tasks.append(self.queue.get_nowait())
            except queue.Empty:
                break
        with self.store.connection() as db:
            for key, url in db.execute(
                    "SELECT id,url FROM projects WHERE detail_status='pending' AND url<>'' "
                    "ORDER BY created_at ASC LIMIT 50"):
                tasks.append((key, url))
        for key, url in tasks:
            if self.stop.is_set():
                break
            host = urlsplit(url).hostname or ''
            if host in self.host_cooldowns and self.host_cooldowns[host] > time.time():
                continue
            try:
                data = self.fetch(url)
                if data:
                    self.store.set_detail(key, data)
            except Exception as error:
                self.store.set_detail(key, error=str(error)[:500])
                if self._blocked(error) and host:
                    self.host_cooldowns[host] = time.time() + 60
            self.stop.wait(self.interval)

    def _loop(self):
        while not self.stop.is_set():
            try:
                self.run_once()
            except Exception as error:
                self.store.record_run('详情补抓', 'error', 0, f'补抓线程异常：{type(error).__name__}')
            self.stop.wait(30 if not self._has_work() else 3)

    def _has_work(self):
        if not self.queue.empty():
            return True
        with self.store.connection() as db:
            return db.execute("SELECT 1 FROM projects WHERE detail_status='pending' AND url<>'' LIMIT 1").fetchone() is not None

    def open(self):
        if self.thread and self.thread.is_alive():
            return
        self.stop.clear()
        self.thread = threading.Thread(target=self._loop, daemon=True, name='workbench-detail-fetcher')
        self.thread.start()

    def close(self):
        self.stop.set()
        if self.thread:
            self.thread.join(timeout=2)
```

```python
"""Backfill win_company/win_amount for existing result notices (idempotent)."""
import sys
from pathlib import Path

from apps.workbench.bidmonitor_workbench.collectors.detail import extract_fields
from apps.workbench.bidmonitor_workbench.paths import database_path
from apps.workbench.bidmonitor_workbench.storage.workbench import WorkbenchStore


def backfill(store, limit=None):
    updated = 0
    while True:
        with store.connection() as db:
            rows = db.execute(
                "SELECT id,content FROM projects WHERE notice_type='中标结果' AND win_company='' AND content<>'' LIMIT 500"
            ).fetchall()
        if not rows:
            break
        for row in rows:
            fields = extract_fields(row['content'])
            win = {k: v for k, v in fields.items() if k in ('win_company', 'win_amount') and v}
            if win:
                store.set_detail(row['id'], win)
                updated += 1
        if limit and updated >= limit:
            break
    return updated


if __name__ == '__main__':
    store = WorkbenchStore(sys.argv[1] if len(sys.argv) > 1 else database_path())
    print(f"回填 {backfill(store)} 条中标公告的 win 字段")
```
